```sql
    data_name2   AS position,
    data_name3   AS product,
    data_option1 AS density
FROM sigong_data
WHERE data_type1 = '제품'
ORDER BY car_type, nation, position, product, density
LIMIT 200;
```

#### (2) 전면 + IRIS BN100 용 농도만 확인

```sql
SELECT 
    data_name3   AS product,
    data_option1 AS density
FROM sigong_data
WHERE data_type1 = '제품'
  AND data_name2 = '전면'
  AND data_name3 = 'IRIS BN100'
ORDER BY data_option1;
```
